cmake_minimum_required(VERSION 3.21.1)
project(DockpipeSDKWindowsProcessFixture LANGUAGES CXX)

if(NOT WIN32)
    message(FATAL_ERROR "The Windows process fixture requires a native Windows target")
endif()
foreach(required DOCKPIPE_SOURCE_SDK_DESCRIPTOR DOCKPIPE_SOURCE_SDK_SHA256)
    if(NOT DEFINED ${required} OR "${${required}}" STREQUAL "")
        message(FATAL_ERROR "${required} must be supplied explicitly")
    endif()
endforeach()
if(NOT IS_ABSOLUTE "${DOCKPIPE_SOURCE_SDK_DESCRIPTOR}")
    message(FATAL_ERROR "Select an absolute installed Dockpipe SDK descriptor")
endif()
file(SHA256 "${DOCKPIPE_SOURCE_SDK_DESCRIPTOR}" actual_source_sdk)
if(NOT actual_source_sdk STREQUAL DOCKPIPE_SOURCE_SDK_SHA256)
    message(FATAL_ERROR "Selected Dockpipe SDK descriptor identity does not match")
endif()
get_filename_component(source_sdk_root "${DOCKPIPE_SOURCE_SDK_DESCRIPTOR}" DIRECTORY)

add_executable(sdk_windows_process_fixture ProcessFixture.cpp
    "${source_sdk_root}/native/WindowsFiles.cpp")
target_include_directories(sdk_windows_process_fixture PRIVATE "${source_sdk_root}/native")
target_compile_features(sdk_windows_process_fixture PRIVATE cxx_std_20)
target_compile_definitions(sdk_windows_process_fixture PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00 NOMINMAX)
if(MINGW)
    target_link_options(sdk_windows_process_fixture PRIVATE -municode)
endif()
