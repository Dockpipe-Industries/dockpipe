cmake_minimum_required(VERSION 3.21.1)
project(LauncherSDKConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTORCC ON)
find_package(DockPipeLauncherSDK 1.0.0 EXACT CONFIG REQUIRED)
add_executable(launcher-sdk-consumer main.cpp ${DockPipeLauncherSDK_TEST_RESOURCES})
target_link_libraries(launcher-sdk-consumer PRIVATE DockPipe::LauncherUI DockPipe::HostBrowser)
configure_file("${DockPipeLauncherSDK_RESOURCE_DIR}/dockpipe-syntax.rcc"
    "${CMAKE_CURRENT_BINARY_DIR}/dockpipe-syntax.rcc" COPYONLY)
