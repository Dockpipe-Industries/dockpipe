cmake_minimum_required(VERSION 3.20)
project(DockpipeInstalledDisplayConsumer LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(DockPipeLauncherSDK 1.0.0 EXACT REQUIRED)
# The public interface header comes solely from the installed candidate SDK.
# No include directory from the host source checkout is permitted.
try_run(abi_result abi_compiled "${CMAKE_CURRENT_BINARY_DIR}/abi-probe"
    "${CMAKE_CURRENT_SOURCE_DIR}/BuildAbi.cpp"
    LINK_LIBRARIES Qt6::Core RUN_OUTPUT_VARIABLE build_abi COMPILE_OUTPUT_VARIABLE abi_compile_output)
if(NOT abi_compiled OR NOT abi_result EQUAL 0)
    message(FATAL_ERROR "Could not read the selected Qt kit build ABI: ${abi_compile_output} ${build_abi}")
endif()
string(STRIP "${build_abi}" build_abi)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
    message(FATAL_ERROR "This disposable consumer proof is Linux x86_64 only")
endif()
foreach(variant IN ITEMS good wrong-qt wrong-compiler wrong-abi wrong-iid)
    set(fixture_qt "${Qt6_VERSION}")
    set(fixture_build_abi "${build_abi}")
    set(fixture_iid "com.dockpipe.launcher.DisplayPackage/1.0")
    set(fixture_compiler "${CMAKE_CXX_COMPILER_VERSION}")
    if(variant STREQUAL "wrong-qt")
        set(fixture_qt "0.0.0")
    elseif(variant STREQUAL "wrong-compiler")
        set(fixture_compiler "incompatible")
    elseif(variant STREQUAL "wrong-abi")
        set(fixture_build_abi "incompatible")
    elseif(variant STREQUAL "wrong-iid")
        set(fixture_iid "com.dockpipe.launcher.DisplayPackage/999.0")
    endif()
    configure_file(metadata.json.in "${variant}.json" @ONLY)
    add_library(display-${variant} MODULE FixtureDisplay.cpp)
    target_compile_definitions(display-${variant} PRIVATE FIXTURE_METADATA="${variant}.json" FIXTURE_IID="${fixture_iid}")
    target_include_directories(display-${variant} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(display-${variant} PRIVATE DockPipe::LauncherHeaders Qt6::Widgets)
    set_target_properties(display-${variant} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/package/assets")
endforeach()
