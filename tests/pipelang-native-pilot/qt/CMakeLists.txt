cmake_minimum_required(VERSION 3.16)
project(PipeLangQtPilot LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Widgets)
if(NOT IS_ABSOLUTE "${PIPELANG_PILOT_INPUT}")
  message(FATAL_ERROR "PIPELANG_PILOT_INPUT must name the generated pilot directory")
endif()
add_executable(pipelang-qt-pilot main.cpp bridge.hpp)
target_include_directories(pipelang-qt-pilot PRIVATE "${PIPELANG_PILOT_INPUT}")
target_compile_options(pipelang-qt-pilot PRIVATE -O2 -g -march=x86-64 -mtune=generic)
target_link_libraries(pipelang-qt-pilot PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
