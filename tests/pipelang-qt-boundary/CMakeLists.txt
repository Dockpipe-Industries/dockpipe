cmake_minimum_required(VERSION 3.16)
project(PipeLangQtBoundary LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core)
if(NOT IS_ABSOLUTE "${PIPELANG_PILOT_INPUT}")
  message(FATAL_ERROR "PIPELANG_PILOT_INPUT must name the admitted generated binding directory")
endif()
add_executable(pipelang-qt-boundary boundary_test.cpp boundary.hpp)
target_include_directories(pipelang-qt-boundary PRIVATE "${PIPELANG_PILOT_INPUT}")
target_compile_options(pipelang-qt-boundary PRIVATE -O2 -g -march=x86-64 -mtune=generic)
target_link_libraries(pipelang-qt-boundary PRIVATE Qt6::Core)
